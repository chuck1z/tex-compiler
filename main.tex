\section*{Part C \& D: 3-Digit Subtraction — Column Method vs. Making 10s}
\vspace{0.5em}

% Example Section
{\Large \textbf{Example: $642 - 285$}}
\vspace{0.5em}

\begin{description}[leftmargin=2em]

  \item[\stepnum{1}{sd1}] \textbf{Make a friendly number for the subtracted term}\\[0.2em]
  Add $+15$ to $285$ to turn it into a clean hundred ($300$).
  \[
    285 + 15 = 300
  \]
  \vspace{0.6em}

  \item[\stepnum{2}{sd2}] \textbf{Add the same amount ($+15$) to the starting number}\\[0.4em]
  Keep the difference the same by adding $+15$ to $642$:
  \[
    642 + 15 = 657
  \]
  \vspace{0.6em}

  \item[\stepnum{3}{sd3}] \textbf{Subtract the friendly number}\\[0.4em]
  Now perform the simplified subtraction:
  \[
    657 - \mathbf{300} = \mathbf{357}
  \]

\end{description}
