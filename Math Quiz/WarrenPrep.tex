\section*{Part 1: Decomposing by Place Value (Split Strategy)}
% ================= EXAMPLE SECTION =================
{\Large \textbf{Example: $6,785 - 2,341$}}
\vspace{0.5em}

\begin{description}[leftmargin=2em]

  \item[\stepnum{1}{p1s1}] \textbf{Decompose the subtrahend into place values}\\[0.2em]
  Break the second number into thousands, hundreds, tens, and ones:
  \begin{itemize}
    \item $2,341 = 2,000 + 300 + 40 + 1$
  \end{itemize}
  \vspace{0.6em}

  \item[\stepnum{2}{p1s2}] \textbf{Subtract the thousands}\\[0.4em]
  Subtract $2,000$ from the starting total ($6,785$):
  \[
    6,785 - 2,000 = 4,785
  \]
  \vspace{0.6em}

  \item[\stepnum{3}{p1s3}] \textbf{Subtract the hundreds}\\[0.4em]
  Subtract $300$ from the running total ($4,785$):
  \[
    4,785 - 300 = 4,485
  \]
  \vspace{0.6em}

  \item[\stepnum{4}{p1s4}] \textbf{Subtract the tens}\\[0.4em]
  Subtract $40$ from the running total ($4,485$):
  \[
    4,485 - 40 = 4,445
  \]
  \vspace{0.6em}

  \item[\stepnum{5}{p1s5}] \textbf{Subtract the remaining ones}\\[0.4em]
  Take away $1$ from $4,445$ to get the final answer:
  \[
    4,445 - \mathbf{1} = \mathbf{4,444}
  \]

\end{description}

\section*{Part 2: Compensation Strategy (Subtracting Near Thousands)}
% ================= EXAMPLE SECTION =================
{\Large \textbf{Example: $5,423 - 1,998$}}
\vspace{0.5em}

\begin{description}[leftmargin=2em]

  \item[\stepnum{1}{p2s1}] \textbf{Round the subtrahend up to the nearest thousand}\\[0.2em]
  $1,998$ is close to $2,000$ (it is $2$ less than $2,000$):
  \[
    1,998 = 2,000 - \mathbf{2}
  \]
  \vspace{0.6em}

  \item[\stepnum{2}{p2s2}] \textbf{Subtract the rounded thousand}\\[0.4em]
  Subtract $2,000$ from $5,423$:
  \[
    5,423 - 2,000 = 3,423
  \]
  \vspace{0.6em}

  \item[\stepnum{3}{p2s3}] \textbf{Add back the difference (Compensation)}\\[0.4em]
  Since you subtracted $2$ too much, add $2$ back to the result:
  \[
    3,423 + \mathbf{2} = \mathbf{3,425}
  \]

\end{description}
